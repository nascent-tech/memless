import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
	<Card title="Fixtures you can read" icon="document">
		Test data lives in a YAML file you can review and diff in Git. Memless reads value types, row identities and relations from the data itself.
	</Card>
	<Card title="The SQL you already know" icon="seti:db">
		`SELECT`, `WHERE`, `JOIN`, `ORDER BY`, `INSERT`, `UPDATE`, `DELETE` and transactions. Anything outside the subset is refused, never approximated.
	</Card>
	<Card title="Writes go back to the file" icon="pencil">
		Every accepted change rewrites the file atomically, so a crash never leaves it half-written.
	</Card>
	<Card title="One engine, three languages" icon="puzzle">
		PHP, Go and Node.js share the same native engine: same file, same SQL, same result, same error message.
	</Card>
</CardGrid>
